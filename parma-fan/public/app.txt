async function api(path) {
  const r = await fetch(path);
  let d = null;
  try { d = await r.json(); } catch {}
  if (!r.ok) throw new Error((d && d.error) || `Ошибка ${r.status}`);
  return d;
}

const cache = {};
function load(key, path, ttl = 60_000) {
  const c = cache[key];
  if (c && Date.now() - c.t < ttl) return c.p;
  const p = api(path).catch((e) => { delete cache[key]; throw e; });
  cache[key] = { p, t: Date.now() };
  return p;
}

function countUp(root = document) {
  for (const el of $$('[data-count]', root)) {
    const target = Number(el.dataset.count);
    if (!Number.isFinite(target)) continue;
    const dec = Number(el.dataset.dec || 0);
    const dur = 900;
    const t0 = performance.now();
    const step = (t) => {
      const k = Math.min(1, (t - t0) / dur);
      const e = 1 - Math.pow(1 - k, 3);
      el.textContent = (target * e).toLocaleString('ru-RU', { minimumFractionDigits: dec, maximumFractionDigits: dec });
      if (k < 1) requestAnimationFrame(step);
    };
    requestAnimationFrame(step);
  }
}

// ── Countdown ──────────────────────────────────────────────────────────────

function countdownHtml() {
  const cell = (k, label) => `<div class="cd-cell"><div class="cd-digits" data-cd="${k}"><div class="flip"><span>0</span></div><div class="flip"><span>0</span></div></div><small>${label}</small></div>`;
  return `<div class="cd">${cell('d', 'дней')}<div class="cd-colon">:</div>${cell('h', 'часов')}<div class="cd-colon">:</div>${cell('m', 'минут')}<div class="cd-colon">:</div>${cell('s', 'секунд')}</div>`;
}
function startCountdown(root, ts) {
  const tick = () => {
    let s = Math.max(0, Math.floor((ts - Date.now()) / 1000));
    const v = { d: Math.min(99, Math.floor(s / 86400)), h: Math.floor((s % 86400) / 3600), m: Math.floor((s % 3600) / 60), s: s % 60 };
    for (const k of Object.keys(v)) {
      const box = $(`[data-cd="${k}"]`, root);
      if (!box) continue;
      const str = String(v[k]).padStart(2, '0');
      $$('.flip', box).forEach((f, i) => {
        const span = f.firstElementChild;
        if (span.textContent !== str[i]) {
          span.textContent = str[i];
          f.classList.remove('go'); void f.offsetWidth; f.classList.add('go');
        }
      });
    }
  };
  tick();
  const id = setInterval(tick, 1000);
  onLeave(() => clearInterval(id));
  return id;
}

async function updateSide() {

    const { live, focus } = await getLive();

      box.innerHTML = `<a href="#/live"><div class="sl-k"><span class="pulse"></span>${live.pregame ? 'Разминка · скоро старт' : `Онлайн · ${esc(live.periodLabel || '')} ${esc(live.clock)}`}</div>
        <div class="sl-row"><span class="sl-team">${teamImg(a, 22)}<span>${esc(shortName(a.name))}</span></span><span class="sl-score">${a.score}</span></div>
        <div class="sl-row"><span class="sl-team">${teamImg(b, 22)}<span>${esc(shortName(b.name))}</span></span><span class="sl-score">${b.score}</span></div></a>`;
      return;
    }
    const g = focus && focus.game;
    if (g) {
      const left = g.ts - Date.now();
      const d = Math.floor(left / 86400000), h = Math.floor((left % 86400000) / 3600000), m = Math.floor((left % 3600000) / 60000);
      box.innerHTML = `<a href="#/live"><div class="sl-k">${left > 0 ? 'Следующий матч' : 'Матч начинается'}</div>
        <div class="sl-row"><span class="sl-team">${teamImg(g.home, 22)}<span>${esc(shortName(g.home.name))}</span></span></div>
        <div class="sl-row"><span class="sl-team">${teamImg(g.away, 22)}<span>${esc(shortName(g.away.name))}</span></span></div>
        <div class="sl-cd">${left > 0 ? `${d ? d + 'д ' : ''}${String(h).padStart(2, '0')}ч ${String(m).padStart(2, '0')}м` : 'скоро старт'}</div></a>`;
    } else box.innerHTML = '';
  } catch { box.innerHTML = ''; }
}

  ]);
  const games = sched.games;
  const finished = games.filter((g) => g.status === 1);
  const last5 = finished.slice(-5);
  const upcoming = games.filter((g) => g.status === 0 && g.ts > Date.now() - 3 * 3600e3).slice(0, 4);
  const main = standings && standings.comps.find((c) => c.id === sched.mainCompId);
  const table = main && main.stages.find((s) => s.type === 'table');
  const usRow = table && table.table.find((r) => isUs(r.teamId));
  const homeFinished = finished.filter((g) => g.isHome && g.attendance);
  const avgAtt = homeFinished.length ? homeFinished.reduce((s, g) => s + g.attendance, 0) / homeFinished.length : null;
  const mainFinished = finished.filter((g) => g.compId === sched.mainCompId);
  const allowed = mainFinished.length ? mainFinished.reduce((s, g) => s + g.opponent.score, 0) / mainFinished.length : null;
  const wins = finished.filter((g) => g.result === 'W').length;

  const kpis = `<section class="panel kpis">

    <div class="kpi"><span class="label">Баланс сезона</span><b><span data-count="${wins}">0</span><small>–${finished.length - wins}</small></b><p>все турниры, ${finished.length} матчей</p></div>

function gameRow(g) {
  const res = g.status === 1 ? `<span class="wl ${g.result}">${g.result === 'W' ? 'В' : 'П'}</span> ${g.home.score}:${g.away.score}` : g.status === 2 ? '<span class="pulse"></span> LIVE' : g.hasTime ? g.timeMsk : '—';
  return `<div class="gl" data-game="${g.id}">
    <div class="gd"><b>${fDay.format(new Date(g.ts))}</b><small>${esc(fMonShort.format(new Date(g.ts)).replace('.', ''))}</small></div>
    <div><div class="go">${teamImg(g.opponent, 28)}${g.isHome ? '' : '@ '}${esc(shortName(g.opponent.name))}</div>
      <div class="gm">${esc(fWeekday.format(new Date(g.ts)))} · ${g.isHome ? 'дома' : 'в гостях'} · ${esc(g.league)}</div></div>
    <div class="gr">${res}</div></div>`;
}

function wireCommon(root, sched, team) {
  $$('[data-game]', root).forEach((el) => el.addEventListener('click', () => {
    const g = sched && sched.games.find((x) => x.id === Number(el.dataset.game));
    if (g) openGame(g);
  }));
  $$('[data-player]', root).forEach((el) => el.addEventListener('click', () => {
    const p = team && team.players.find((x) => x.id === Number(el.dataset.player));
    if (p) openPlayer(p, team);
  }));

function wormSvg(prog, teams) {
  if (!prog || prog.length < 2) return '<div class="empty">График появится после первых очков</div>';
  const W = 900, H = 180, P = 26;
  const tMax = Math.max(40, prog[prog.length - 1].t);
  const diffs = prog.map((p) => p.a - p.b);
  const m = Math.max(8, ...diffs.map(Math.abs));
  const x = (t) => P + (t / tMax) * (W - P * 2);
  const y = (d) => H / 2 - (d / m) * (H / 2 - 16);
  let path = `M${x(0)},${y(0)}`;
  let prev = 0;
  for (const p of prog) { path += ` L${x(p.t).toFixed(1)},${y(prev).toFixed(1)} L${x(p.t).toFixed(1)},${y(p.a - p.b).toFixed(1)}`; prev = p.a - p.b; }
  const area = `${path} L${x(prog[prog.length - 1].t)},${H / 2} L${x(0)},${H / 2} Z`;
  const qs = [10, 20, 30, 40, 45, 50].filter((q) => q <= tMax);
  const usTop = isUs(teams[0].id);
  return `<svg class="worm" viewBox="0 0 ${W} ${H}" preserveAspectRatio="none">
    <defs><clipPath id="up"><rect x="0" y="0" width="${W}" height="${H / 2}"/></clipPath><clipPath id="dn"><rect x="0" y="${H / 2}" width="${W}" height="${H / 2}"/></clipPath></defs>
    <line class="axis" x1="${P}" x2="${W - P}" y1="${H / 2}" y2="${H / 2}"/>
    ${qs.map((q) => `<line class="qline" x1="${x(q)}" x2="${x(q)}" y1="8" y2="${H - 8}"/><text x="${x(q) - 4}" y="${H - 2}" text-anchor="end">${q}'</text>`).join('')}
    <path d="${area}" clip-path="url(#up)" fill="${usTop ? 'rgba(255,81,0,.28)' : 'rgba(46,91,216,.35)'}" stroke="none"/>
    <path d="${area}" clip-path="url(#dn)" fill="${usTop ? 'rgba(46,91,216,.35)' : 'rgba(255,81,0,.28)'}" stroke="none"/>
    <path class="draw" d="${path}" stroke="#EEF2FA" stroke-width="1.6" fill="none" vector-effect="non-scaling-stroke"/>
    <text x="${P}" y="14">${esc(shortName(teams[0].name))} ▲</text><text x="${P}" y="${H - 14}">${esc(shortName(teams[1].name))} ▼</text>
  </svg>`;
}

    ${t.players.map((p) => `<tr><td><span class="nm">${esc(p.num)}</span>${esc(p.name)}</td><td class="pts">${p.pts}</td><td>${esc(p.fg2 || '—')}</td><td>${esc(p.fg3 || '—')}</td><td>${esc(p.ft || '—')}</td><td>${p.reb ?? 0}</td>${hasAst ? `<td>${p.ast ?? 0}</td>` : ''}<td>${p.stl ?? 0}</td><td>${p.blk ?? 0}</td><td>${p.pf ?? 0}</td>${t.players.some((x) => x.min) ? `<td>${esc(p.min || '—')}</td><td>${p.pm ?? '—'}</td>` : ''}</tr>`).join('')}

function feedHtml(live) {
  if (!live.feed.length) return '<div class="empty">События появятся после стартового вбрасывания</div>';
  return live.feed.map((f) => {
    const t = live.teams[f.team - 1];
    const c = isUs(t.id) ? 'var(--orange)' : 'var(--royal)';
    return `<div class="feed-item ${f.pts ? 'score' : ''}"><span class="ft">${f.period <= 4 ? f.period + 'Ч' : 'ОТ'} ${esc(f.clock)}</span>
      <span class="fp" style="--c:${c}">${f.player ? esc(f.player) + ' — ' : esc(shortName(t.name)) + ' — '}${f.pts ? `<b>${esc(f.text)}</b>` : esc(f.text)}</span>
      <span class="fs">${f.a}:${f.b}</span></div>`;
  }).join('');
}

  };

  let clockSec = live ? live.clockSec : 0;
  let clockAt = Date.now();
  const tickId = setInterval(() => {
    if (!live || !live.clockRunning) return;
    const s = Math.max(0, clockSec - Math.floor((Date.now() - clockAt) / 1000));
    const el = $('#clock');
    if (el) el.textContent = `${String(Math.floor(s / 60)).padStart(2, '0')}:${String(s % 60).padStart(2, '0')}`;

        <div class="bl-game ${g.id === gameId ? 'on' : ''} ${isUs(g.home.id) || isUs(g.away.id) ? 'us' : ''}" data-id="${g.id}">

      $$('.bl-game', box).forEach((el) => (el.onclick = () => (location.hash = '#/live/' + el.dataset.id)));
      $('#boardUpd').textContent = 'обновлено ' + dtf({ hour: '2-digit', minute: '2-digit', second: '2-digit' }).format(new Date());
    } catch {}
  };
  paintBoard();

  const poll = async () => {
    try {
      const r = gameId ? { live: (await getLive(gameId)).live } : await getLive();
      if (!r.live) {
        if (!live) return;
      }
      if (!live && r.live) { route(); return; }
      if (!r.live) return;
      const prev = live;
      live = r.live;
      clockSec = live.clockSec; clockAt = Date.now();
      const board = $('#board');
      if (board) {
        board.outerHTML = boardHtml(live);
        live.teams.forEach((t, i) => {
          if (prev && prev.teams[i].score !== t.score) {

      paintTab();
    } catch {}
    paintBoard();
  };
  const isActive = live ? live.status === 2 : focus && focus.game && focus.game.ts - Date.now() < 30 * 60e3;
  const pid = setInterval(poll, isActive ? 10_000 : 60_000);
  onLeave(() => clearInterval(pid));
}

const MONTHS = ['Январь', 'Февраль', 'Март', 'Апрель', 'Май', 'Июнь', 'Июль', 'Август', 'Сентябрь', 'Октябрь', 'Ноябрь', 'Декабрь'];
const calState = { y: null, m: null, comp: 'all' };

async function renderCalendar(view) {
  const sched = await getSchedule();
  if (calState.y == null) {
    const next = sched.games.find((g) => g.status !== 1 && g.ts > Date.now() - 86400e3) || sched.games[sched.games.length - 1];
    const p = mskParts(next ? next.ts : Date.now());
    calState.y = p.y; calState.m = p.m;
  }
  const paint = () => {
    const games = sched.games.filter((g) => calState.comp === 'all' || g.compId === Number(calState.comp));
    const byDay = {};
    for (const g of games) (byDay[dayKey(g.ts)] ||= []).push(g);
    const first = new Date(Date.UTC(calState.y, calState.m - 1, 1));
    const startDow = (first.getUTCDay() + 6) % 7;
    const daysIn = new Date(Date.UTC(calState.y, calState.m, 0)).getUTCDate();
    const today = dayKey(Date.now());
    const cells = [];
    const total = Math.ceil((startDow + daysIn) / 7) * 7;
    for (let i = 0; i < total; i++) {
      const d = new Date(Date.UTC(calState.y, calState.m - 1, 1 + i - startDow));
      const key = d.toISOString().slice(0, 10);
      const out = d.getUTCMonth() !== calState.m - 1;
      const gs = byDay[key] || [];
      cells.push(`<div class="cal-day ${out ? 'out' : ''} ${key === today ? 'today' : ''}"><span class="d">${d.getUTCDate()}</span>
        ${gs.map((g) => `<div class="cal-game ${g.isHome ? 'home' : ''}" data-game="${g.id}"><div class="cg-t">${teamImg(g.opponent, 18)}<span>${g.isHome ? '' : '@'}${esc(shortName(g.opponent.name))}</span></div>
          <div class="cg-s"><span>${g.status === 1 ? `<span class="${g.result}">${g.result === 'W' ? 'В' : 'П'}</span> ${g.home.score}:${g.away.score}` : g.status === 2 ? '● LIVE' : g.hasTime ? g.timeMsk : '—'}</span><span>${esc(g.league.split(' ')[0])}</span></div></div>`).join('')}
      </div>`);
    }
    const monthGames = games.filter((g) => { const p = mskParts(g.ts); return p.y === calState.y && p.m === calState.m; });
    view.innerHTML = `<div class="toolbar"><div class="chips" id="compChips">
        <button class="chip ${calState.comp === 'all' ? 'on' : ''}" data-comp="all">Все турниры</button>
        ${sched.comps.map((c) => `<button class="chip ${String(calState.comp) === String(c.id) ? 'on' : ''}" data-comp="${c.id}">${esc(c.name)}</button>`).join('')}
      </div><a class="btn btn-sm" href="/api/calendar.ics">${icon('i-plus')}Весь сезон в мой календарь (.ics)</a></div>
      <div class="cal-wrap">
        <section class="panel">
          <div class="cal-head"><h2>${MONTHS[calState.m - 1]} <span class="muted">${calState.y}</span></h2>
            <div class="cal-nav"><button class="icon-btn" id="calPrev" aria-label="Назад">${icon('i-left')}</button><button class="btn btn-sm" id="calToday">Сегодня</button><button class="icon-btn" id="calNext" aria-label="Вперёд">${icon('i-right')}</button></div></div>
          <div class="cal-grid">${['Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб', 'Вс'].map((d) => `<div class="cal-dow">${d}</div>`).join('')}${cells.join('')}</div>
          <div class="cal-legend"><span><i class="h"></i>домашний матч · УДС «Молот»</span><span><i></i>выезд</span><span>Время — московское</span></div>
        </section>
        <section class="panel"><div class="panel-head"><h2>Матчи месяца</h2><span class="label">${monthGames.length}</span></div>
          <div class="glist">${monthGames.map(gameRow).join('') || '<div class="empty">В этом месяце матчей нет</div>'}</div></section>
      </div>`;
    $$('#compChips .chip', view).forEach((b) => (b.onclick = () => { calState.comp = b.dataset.comp; paint(); }));
    $('#calPrev').onclick = () => { calState.m--; if (calState.m < 1) { calState.m = 12; calState.y--; } paint(); };
    $('#calNext').onclick = () => { calState.m++; if (calState.m > 12) { calState.m = 1; calState.y++; } paint(); };
    $('#calToday').onclick = () => { const p = mskParts(Date.now()); calState.y = p.y; calState.m = p.m; paint(); };
    wireCommon(view, sched, null);
    $$('#view > *').forEach((el, i) => el.style.setProperty('--i', i));
  };
  paint();
}

function plural(n, a, b, c) { n = Math.abs(n) % 100; const n1 = n % 10; if (n > 10 && n < 20) return c; if (n1 > 1 && n1 < 5) return b; if (n1 === 1) return a; return c; }

function tilt(root) {
  for (const card of $$('.card', root)) {
    card.addEventListener('pointermove', (e) => {
      const r = card.getBoundingClientRect();
      const px = (e.clientX - r.left) / r.width, py = (e.clientY - r.top) / r.height;
      card.style.setProperty('--ry', `${(px - 0.5) * 10}deg`);
      card.style.setProperty('--rx', `${(0.5 - py) * 10}deg`);
      card.style.setProperty('--mx', `${px * 100}%`);
      card.style.setProperty('--my', `${py * 100}%`);
    });
    card.addEventListener('pointerleave', () => { card.style.setProperty('--rx', '0deg'); card.style.setProperty('--ry', '0deg'); });
  }
}

const toRad = (d) => (d * Math.PI) / 180;
function km(a, b) {
  const R = 6371, dLat = toRad(b[0] - a[0]), dLon = toRad(b[1] - a[1]);
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(toRad(a[0])) * Math.cos(toRad(b[0])) * Math.sin(dLon / 2) ** 2;
  return Math.round(2 * R * Math.asin(Math.sqrt(h)));
}

function mdLite(s) {
  const lines = esc(s).replace(/\*\*(.+?)\*\*/g, '<strong>$1</strong>').split('\n');
  let html = '', inList = false;
  for (const raw of lines) {
    const l = raw.trim();
    if (/^[-•*]\s+/.test(l)) { if (!inList) { html += '<ul>'; inList = true; } html += `<li>${l.replace(/^[-•*]\s+/, '')}</li>`; continue; }
    if (inList) { html += '</ul>'; inList = false; }
    if (l) html += `<p>${l.replace(/^#+\s*/, '')}</p>`;
  }
  if (inList) html += '</ul>';
  return html;
}

function chatHtml(withHello = true) {
  return `<div class="chat-log"></div><div class="chat-sugg"></div>
    <form class="chat-form"><textarea rows="1" placeholder="Спросите про матч, игрока, билеты или баскетбол…"></textarea><button type="submit" aria-label="Отправить">${icon('i-send')}</button></form>`;
}

function mountChat(root) {
  root.innerHTML = chatHtml();
  const v = { root, log: $('.chat-log', root), sugg: $('.chat-sugg', root), form: $('.chat-form', root), ta: $('textarea', root) };
  chat.views.add(v);
  v.form.onsubmit = (e) => { e.preventDefault(); ask(v.ta.value); v.ta.value = ''; v.ta.style.height = ''; };
  v.ta.addEventListener('keydown', (e) => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); v.form.requestSubmit(); } });
  v.ta.addEventListener('input', () => { v.ta.style.height = 'auto'; v.ta.style.height = Math.min(160, v.ta.scrollHeight) + 'px'; });
  paintChat(v);
  return v;
}

function paintChat(v) {
  const hello = `<div class="chat-hello"><h3>Привет, я Пармыч</h3><p>Знаю расписание, состав, статистику, таблицы и счёт идущего матча — всё из официального протокола лиги. Спрашивайте что угодно: от «кто сегодня набрал больше всех» до «как работает фол в нападении».</p></div>`;
  v.log.innerHTML = (chat.messages.length ? '' : hello) + chat.messages.map((m) => `<div class="msg ${m.role === 'user' ? 'user' : 'bot'}">${m.role === 'user' ? esc(m.content) : m.content ? mdLite(m.content) : '<span class="typing"><i></i><i></i><i></i></span>'}</div>`).join('');
  v.sugg.innerHTML = chat.messages.length ? '' : SUGGEST.map((q) => `<button class="chip" type="button">${esc(q)}</button>`).join('');
  $$('.chip', v.sugg).forEach((b) => (b.onclick = () => ask(b.textContent)));
  $('button[type=submit]', v.form).disabled = chat.busy;
  v.log.scrollTop = v.log.scrollHeight;
}
const paintAll = () => { for (const v of chat.views) if (document.body.contains(v.root)) paintChat(v); else chat.views.delete(v); };

async function ask(q) {
  q = (q || '').trim();
  if (!q || chat.busy) return;
  chat.busy = true;
  chat.messages.push({ role: 'user', content: q }, { role: 'assistant', content: '' });
  paintAll();
  const bot = chat.messages[chat.messages.length - 1];
  try {
    const r = await fetch('/api/ai', { method: 'POST', headers: { 'content-type': 'application/json' }, body: JSON.stringify({ messages: chat.messages.slice(0, -1) }) });
    if (!r.ok || !r.body) { const e = await r.json().catch(() => ({})); throw new Error(e.error || 'ошибка ' + r.status); }
    const reader = r.body.getReader();
    const dec = new TextDecoder();
    let buf = '', last = 0;
    while (true) {
      const { done, value } = await reader.read();
      if (done) break;
      buf += dec.decode(value, { stream: true });
      const parts = buf.split('\n\n');
      buf = parts.pop();
      for (const p of parts) {
        const line = p.trim();
        if (!line.startsWith('data:')) continue;
        const payload = line.slice(5).trim();
        if (payload === '[DONE]') continue;
        try { const ev = JSON.parse(payload); if (ev.t) bot.content += ev.t; if (ev.error && !bot.content) bot.content = 'Не получилось ответить: ' + ev.error; } catch {}
      }
      if (performance.now() - last > 60) { last = performance.now(); paintAll(); }
    }
    if (!bot.content) bot.content = 'Хм, ответ пустой. Попробуйте переформулировать вопрос.';
  } catch (e) {
    bot.content = 'Пармыч временно недоступен: ' + e.message;
  }
  chat.busy = false;
  try { sessionStorage.setItem('parmych', JSON.stringify(chat.messages.slice(-20))); } catch {}
  paintAll();
}
